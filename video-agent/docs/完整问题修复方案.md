### 修复 4: 大文件分页读取

**文件：** `scripts/analyze-large-file.mjs`

```javascript
async function readLargeFile(filePath, maxTokens = 25000) {
  const stats = await fs.stat(filePath);
  const estimatedLines = Math.ceil(stats.size / 80); // 估算行数
  
  const pages = [];
  let offset = 0;
  const linesPerPage = 800;
  
  while (offset < estimatedLines) {
    const content = await Read(filePath, {
      offset,
      limit: linesPerPage
    });
    pages.push(content);
    offset += linesPerPage;
  }
  
  return pages;
}
```

### 修复 5: 素材匹配策略优化

**文件：** `lib/creative/material-matching.mjs`

```javascript
// 智能素材匹配
function matchMaterialsToRequest(request, availableMaterials) {
  const { message, product } = request;
  
  // 检查是否有功效声明
  const hasHealthClaim = /(?:功效|效果|治疗|补充|预防)/u.test(message);
  
  if (hasHealthClaim) {
    console.warn('⚠️  检测到功效声明，需要证据支撑');
    // 建议使用通用场景替代
    return {
      matched: false,
      reason: 'health_claim_requires_evidence',
      suggestion: '建议使用通用展示场景，避免具体功效声明'
    };
  }
  
  // 按相关性匹配素材
  const matched = availableMaterials.filter(material => {
    return material.name.includes(product.name) ||
           material.tags?.includes(product.category);
  });
  
  return {
    matched: matched.length > 0,
    materials: matched,
    suggestion: matched.length === 0 
      ? '建议使用"产品展示"等通用场景'
      : null
  };
}
```
